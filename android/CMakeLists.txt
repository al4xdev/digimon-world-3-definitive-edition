cmake_minimum_required(VERSION 3.20)
project(dw2003_android C)

set(SDL3_ANDROID_DIR "/opt/SDL3" CACHE PATH "Path to unpacked SDL3 source tree")
if(NOT EXISTS "${SDL3_ANDROID_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "SDL3 source tree not found at ${SDL3_ANDROID_DIR}")
endif()

set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_OPENSLES ON CACHE BOOL "" FORCE)
add_subdirectory("${SDL3_ANDROID_DIR}" sdl3_build EXCLUDE_FROM_ALL)

set(PSXSTACK_SDL ON CACHE BOOL "" FORCE)
set(PSXSTACK_SDL2 OFF CACHE BOOL "" FORCE)
if(NOT PSXSTACK_DXC)
    set(PSXSTACK_DXC "${CMAKE_CURRENT_SOURCE_DIR}/../tools/dxc/bin/dxc" CACHE FILEPATH "")
endif()
set(PSXSTACK_OBJCOPY "${CMAKE_OBJCOPY}" CACHE FILEPATH "" FORCE)
set(PSXSTACK_OBJDUMP "${CMAKE_OBJDUMP}" CACHE FILEPATH "" FORCE)

add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../port" dw3_port)
